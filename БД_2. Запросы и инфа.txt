-- Запрос 7.1. Хранимая процедура: добавление нового заказа
-- SQL запрос:
CREATE OR ALTER PROCEDURE sp_PlaceOrder
    @CustomerLastName NVARCHAR(50),
    @BoatModel NVARCHAR(100),
    @Quantity INT,
    @DeliveryAddress NVARCHAR(200)
AS
BEGIN
    DECLARE @CustomerID INT, @BoatID INT, @Price DECIMAL(12,2), @TotalAmount DECIMAL(12,2);
    
    -- Получаем ID клиента
    SELECT @CustomerID = CustomerID FROM Customers WHERE LastName = @CustomerLastName;
    
    -- Получаем ID лодки и цену
    SELECT @BoatID = BoatID, @Price = Price FROM Boats WHERE ModelName = @BoatModel;
    
    -- Проверяем наличие
    IF @CustomerID IS NOT NULL AND @BoatID IS NOT NULL
    BEGIN
        SET @TotalAmount = @Price * @Quantity;
        
        INSERT INTO Orders (CustomerID, BoatID, Quantity, OrderDate, TotalAmount, Status, DeliveryAddress)
        VALUES (@CustomerID, @BoatID, @Quantity, GETDATE(), @TotalAmount, 'В обработке', @DeliveryAddress);
        
        PRINT 'Заказ успешно оформлен на сумму ' + CAST(@TotalAmount AS NVARCHAR(20)) + ' руб.';
    END
    ELSE
    BEGIN
        PRINT 'Ошибка: клиент или лодка не найдены';
    END
END;
GO

-- Запрос 7.2. Хранимая процедура: отчёт по продажам за период
-- SQL запрос:
CREATE OR ALTER PROCEDURE sp_SalesReport
    @StartDate DATE,
    @EndDate DATE
AS
BEGIN
    SELECT 
        b.ModelName,
        bc.CategoryName,
        COUNT(o.OrderID) AS OrdersCount,
        SUM(o.Quantity) AS UnitsSold,
        SUM(o.TotalAmount) AS Revenue,
        AVG(o.TotalAmount) AS AvgOrderValue
    FROM Orders o
    INNER JOIN Boats b ON o.BoatID = b.BoatID
    INNER JOIN BoatCategories bc ON b.CategoryID = bc.CategoryID
    WHERE o.OrderDate BETWEEN @StartDate AND @EndDate
    GROUP BY b.ModelName, bc.CategoryName
    ORDER BY Revenue DESC;
END;
GO

    )
    BEGIN
        RAISERROR('Ошибка: недостаточно товара на складе!', 16, 1);
        RETURN;
    END
    ELSE
    BEGIN
        INSERT INTO Orders (CustomerID, BoatID, OrderDate, Quantity, TotalAmount, Status, DeliveryAddress)
        SELECT CustomerID, BoatID, OrderDate, Quantity, TotalAmount, Status, DeliveryAddress
        FROM inserted;
        
        -- Уменьшаем количество на складе
        UPDATE b
        SET StockQuantity = b.StockQuantity - i.Quantity
        FROM Boats b
        INNER JOIN inserted i ON b.BoatID = i.BoatID;
        
        PRINT 'Заказ добавлен, остаток на складе обновлён';
    END
END;
GO

-- Запрос 8.2. Триггер на UPDATE: логирование изменения статуса заказа
-- SQL запрос:
-- Создаём таблицу для логов
IF NOT EXISTS (SELECT * FROM sys.tables WHERE name = 'OrderStatusLog')
BEGIN
    CREATE TABLE OrderStatusLog (
        LogID INT IDENTITY(1,1) PRIMARY KEY,
        OrderID INT,
        OldStatus NVARCHAR(30),
        NewStatus NVARCHAR(30),
        ChangeDate DATETIME,
        ChangedBy NVARCHAR(100)
    );
END;
GO

CREATE OR ALTER TRIGGER trg_Order_StatusChange
ON Orders
AFTER UPDATE
AS
BEGIN
    IF UPDATE(Status)
    BEGIN
        INSERT INTO OrderStatusLog (OrderID, OldStatus, NewStatus, ChangeDate, ChangedBy)
        SELECT 
            i.OrderID,
            d.Status,
            i.Status,
            GETDATE(),
            SYSTEM_USER
        FROM inserted i
        INNER JOIN deleted d ON i.OrderID = d.OrderID
        WHERE i.Status != d.Status;
        
        PRINT 'Изменение статуса зафиксировано';
    END
END;
GO

-- Запрос 8.3. Триггер на DELETE: запрет удаления клиентов с активными заказами
-- SQL запрос:
CREATE OR ALTER TRIGGER trg_Customer_DeleteCheck
ON Customers
INSTEAD OF DELETE
AS
BEGIN
    IF EXISTS (
        SELECT 1 FROM deleted d
        INNER JOIN Orders o ON d.CustomerID = o.CustomerID
        WHERE o.Status IN ('В обработке', 'Оплачен')
    )
    BEGIN
        RAISERROR('Нельзя удалить клиента с активными заказами!', 16, 1);
        RETURN;
    END
    ELSE
    BEGIN
        DELETE FROM Customers WHERE CustomerID IN (SELECT CustomerID FROM deleted);
        PRINT 'Клиент успешно удалён';
    END
END;
GO

-- Запрос 9.1. Представление: полная информация о заказах
-- SQL запрос:
CREATE OR ALTER VIEW vw_FullOrderInfo AS
SELECT 
    o.OrderID,
    CONCAT(c.LastName, ' ', c.FirstName, ' ', ISNULL(c.MiddleName, '')) AS CustomerName,
    c.Phone,
    c.City,
    b.ModelName AS BoatModel,
    bc.CategoryName,
    m.Name AS Manufacturer,
    o.Quantity,
    b.Price AS UnitPrice,
    o.TotalAmount,
    o.OrderDate,
    o.Status,
    o.DeliveryAddress,
    DATEDIFF(DAY, o.OrderDate, GETDATE()) AS DaysSinceOrder
FROM Orders o
INNER JOIN Customers c ON o.CustomerID = c.CustomerID
INNER JOIN Boats b ON o.BoatID = b.BoatID
INNER JOIN BoatCategories bc ON b.CategoryID = bc.CategoryID
INNER JOIN Manufacturers m ON b.ManufacturerID = m.ManufacturerID;
GO

-- Пример использования:
SELECT * FROM vw_FullOrderInfo WHERE Status = 'Доставлен';
GO

-- Запрос 10.1. Ранжирование клиентов по сумме покупок
-- SQL запрос:
SELECT 
    c.CustomerID,
    CONCAT(c.LastName, ' ', c.FirstName) AS CustomerName,
    c.City,
    COUNT(o.OrderID) AS OrderCount,
    SUM(o.TotalAmount) AS TotalSpent,
    RANK() OVER (ORDER BY SUM(o.TotalAmount) DESC) AS RankBySpent,
    DENSE_RANK() OVER (ORDER BY SUM(o.TotalAmount) DESC) AS DenseRank,
    ROW_NUMBER() OVER (ORDER BY SUM(o.TotalAmount) DESC) AS RowNum,
    AVG(SUM(o.TotalAmount)) OVER () AS AvgSpentAll,
    SUM(SUM(o.TotalAmount)) OVER () AS TotalRevenueAll,
    SUM(o.TotalAmount) * 100.0 / SUM(SUM(o.TotalAmount)) OVER () AS PercentOfTotal
FROM Customers c
LEFT JOIN Orders o ON c.CustomerID = o.CustomerID
GROUP BY c.CustomerID, c.LastName, c.FirstName, c.City
ORDER BY TotalSpent DESC;
GO
