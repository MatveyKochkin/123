-- =====================================================
-- База данных: интернет-магазин лодок "ООО Капитан"
-- СУБД: MySQL
-- =====================================================

-- 1. Таблица: Категории лодок
CREATE TABLE BoatCategories (
    CategoryID INT AUTO_INCREMENT PRIMARY KEY,
    CategoryName VARCHAR(50) NOT NULL UNIQUE,
    Description VARCHAR(200)
);

-- 2. Таблица: Производители лодок
CREATE TABLE Manufacturers (
    ManufacturerID INT AUTO_INCREMENT PRIMARY KEY,
    Name VARCHAR(100) NOT NULL UNIQUE,
    Country VARCHAR(50),
    Website VARCHAR(100)
);

-- 3. Таблица: Лодки
CREATE TABLE Boats (
    BoatID INT AUTO_INCREMENT PRIMARY KEY,
    ModelName VARCHAR(100) NOT NULL,
    CategoryID INT NOT NULL,
    ManufacturerID INT NOT NULL,
    Length DECIMAL(5,2),
    Width DECIMAL(4,2),
    MaxPassengers INT,
    EnginePower INT,
    Price DECIMAL(12,2) NOT NULL,
    StockQuantity INT NOT NULL DEFAULT 0,
    FOREIGN KEY (CategoryID) REFERENCES BoatCategories(CategoryID),
    FOREIGN KEY (ManufacturerID) REFERENCES Manufacturers(ManufacturerID)
);

-- 4. Таблица: Клиенты
CREATE TABLE Customers (
    CustomerID INT AUTO_INCREMENT PRIMARY KEY,
    LastName VARCHAR(50) NOT NULL,
    FirstName VARCHAR(50) NOT NULL,
    MiddleName VARCHAR(50),
    Email VARCHAR(100) UNIQUE NOT NULL,
    Phone VARCHAR(20) NOT NULL,
    City VARCHAR(50),
    RegistrationDate DATE NOT NULL DEFAULT (CURRENT_DATE)
);

-- 5. Таблица: Заказы (ИСПРАВЛЕННАЯ)
CREATE TABLE Orders (
    OrderID INT AUTO_INCREMENT PRIMARY KEY,
    CustomerID INT NOT NULL,
    BoatID INT NOT NULL,
    OrderDate DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    Quantity INT NOT NULL DEFAULT 1,
    TotalAmount DECIMAL(12,2) NOT NULL,
    Status VARCHAR(30) NOT NULL DEFAULT 'В обработке',
    DeliveryAddress VARCHAR(200),
    FOREIGN KEY (CustomerID) REFERENCES Customers(CustomerID),
    FOREIGN KEY (BoatID) REFERENCES Boats(BoatID)
);
