-- Запрос 4.1. Автомобили, у которых сумма штрафов выше среднего (вложенный запрос)
-- SQL запрос:
SELECT v.LicensePlate, SUM(vln.FineAmount) AS TotalFine
FROM TrafficViolations tv
INNER JOIN Vehicles v ON tv.VehicleID = v.VehicleID
INNER JOIN Violations vln ON tv.ViolationID = vln.ViolationID
GROUP BY v.LicensePlate
HAVING SUM(vln.FineAmount) > (
    SELECT AVG(TotalFine)
    FROM (
        SELECT SUM(vln2.FineAmount) AS TotalFine
        FROM TrafficViolations tv2
        INNER JOIN Violations vln2 ON tv2.ViolationID = vln2.ViolationID
        GROUP BY tv2.VehicleID
    ) AS SubQuery
);
GO

-- Запрос 4.2. Сотрудники, которые выписали больше 2 штрафов (с UNION)

SELECT o.LastName, o.FirstName, COUNT(*) AS FinesCount
FROM TrafficViolations tv
INNER JOIN Officers o ON tv.OfficerID = o.OfficerID
GROUP BY o.LastName, o.FirstName
HAVING COUNT(*) > 2

UNION

SELECT 'ALL OTHERS' AS LastName, '' AS FirstName, SUM(FinesCount) AS FinesCount
FROM (
    SELECT COUNT(*) AS FinesCount
    FROM TrafficViolations tv
    INNER JOIN Officers o ON tv.OfficerID = o.OfficerID
    GROUP BY o.LastName, o.FirstName
    HAVING COUNT(*) <= 2
) AS Others;
GO

-- Запрос 7.1. Хранимая процедура: добавление нового нарушения
-- SQL запрос:
CREATE OR ALTER PROCEDURE sp_AddTrafficViolation
    @LicensePlate NVARCHAR(10),
    @OfficerLastName NVARCHAR(50),
    @ViolationCode CHAR(8),
    @ViolationDateTime DATETIME,
    @Location NVARCHAR(200)
AS
BEGIN
    DECLARE @VehicleID INT, @OfficerID INT, @ViolationID INT;
    
    -- Получаем ID автомобиля
    SELECT @VehicleID = VehicleID FROM Vehicles WHERE LicensePlate = @LicensePlate;
    
    -- Получаем ID сотрудника
    SELECT @OfficerID = OfficerID FROM Officers WHERE LastName = @OfficerLastName;
    
    -- Получаем ID нарушения
    SELECT @ViolationID = ViolationID FROM Violations WHERE ViolationCode = @ViolationCode;
    
    -- Проверяем, что все ID найдены
    IF @VehicleID IS NOT NULL AND @OfficerID IS NOT NULL AND @ViolationID IS NOT NULL
    BEGIN
        INSERT INTO TrafficViolations (VehicleID, OfficerID, ViolationID, ViolationDateTime, Location, IsPaid)
        VALUES (@VehicleID, @OfficerID, @ViolationID, @ViolationDateTime, @Location, 0);
        
        PRINT 'Нарушение успешно добавлено!';
    END
    ELSE
    BEGIN
        PRINT 'Ошибка: не найдены данные о машине, сотруднике или нарушении';
    END
END;
GO

-- Запрос 7.2. Хранимая процедура: отчёт по автомобилю
-- SQL запрос:
CREATE OR ALTER PROCEDURE sp_CarViolationReport
    @LicensePlate NVARCHAR(10)
AS
BEGIN
    SELECT 
        v.LicensePlate,
        v.VIN,
        vm.MakeName,
        vmd.ModelName,
        COUNT(tv.TrafficViolationID) AS ViolationsCount,
        SUM(vln.FineAmount) AS TotalFine,
        SUM(CASE WHEN tv.IsPaid = 1 THEN vln.FineAmount ELSE 0 END) AS PaidAmount,
        SUM(CASE WHEN tv.IsPaid = 0 THEN vln.FineAmount ELSE 0 END) AS DebtAmount
    FROM Vehicles v
    LEFT JOIN TrafficViolations tv ON v.VehicleID = tv.VehicleID
    LEFT JOIN Violations vln ON tv.ViolationID = vln.ViolationID
    LEFT JOIN VehicleModels vmd ON v.ModelID = vmd.ModelID
    LEFT JOIN VehicleMakes vm ON vmd.MakeID = vm.MakeID
    WHERE v.LicensePlate = @LicensePlate
    GROUP BY v.LicensePlate, v.VIN, vm.MakeName, vmd.ModelName;
END;
GO

-- Запрос 8.1. Триггер на INSERT: проверка минимального года выпуска

CREATE OR ALTER TRIGGER trg_Vehicle_InsertCheck
ON Vehicles

-- Запрос 8.2. Триггер на UPDATE: логирование изменения оплаты штрафа
-- SQL запрос:
CREATE OR ALTER TRIGGER trg_TrafficViolation_UpdateLog
ON TrafficViolations
AFTER UPDATE
AS
BEGIN
    IF UPDATE(IsPaid)
    BEGIN
        INSERT INTO TrafficViolationsLog (TrafficViolationID, OldIsPaid, NewIsPaid, ChangeDate)
        SELECT 
            i.TrafficViolationID,
            d.IsPaid,
            i.IsPaid,
            GETDATE()
        FROM inserted i
        INNER JOIN deleted d ON i.TrafficViolationID = d.TrafficViolationID
        WHERE i.IsPaid != d.IsPaid;
        
        PRINT 'Изменение оплаты зафиксировано в логе';
    END
END;
GO

-- Создаём таблицу для логов (для триггера UPDATE)
IF NOT EXISTS (SELECT * FROM sys.tables WHERE name = 'TrafficViolationsLog')
BEGIN
    CREATE TABLE TrafficViolationsLog (
        LogID INT IDENTITY(1,1) PRIMARY KEY,
        TrafficViolationID INT,
        OldIsPaid BIT,
        NewIsPaid BIT,
        ChangeDate DATETIME
    );
END;
GO

-- Запрос 8.3. Триггер на DELETE: запрет удаления автомобилей с неоплаченными штрафами
-- SQL запрос:
CREATE OR ALTER TRIGGER trg_Vehicle_DeleteCheck
ON Vehicles
INSTEAD OF DELETE
AS
BEGIN
    IF EXISTS (
        SELECT 1 FROM deleted d
        INNER JOIN TrafficViolations tv ON d.VehicleID = tv.VehicleID
        WHERE tv.IsPaid = 0
    )
    BEGIN
        RAISERROR('Нельзя удалить автомобиль с неоплаченными штрафами!', 16, 1);
        RETURN;
    END
    ELSE
    BEGIN
        DELETE FROM Vehicles WHERE VehicleID IN (SELECT VehicleID FROM deleted);
        PRINT 'Автомобиль успешно удалён';
    END;
END;
GO

-- Запрос 10.1. Ранжирование автомобилей по сумме штрафов
-- SQL запрос:
SELECT 
    v.LicensePlate,
    vm.MakeName,
    vmd.ModelName,
    SUM(vln.FineAmount) AS TotalFine,
    RANK() OVER (ORDER BY SUM(vln.FineAmount) DESC) AS RankByFine,
    DENSE_RANK() OVER (ORDER BY SUM(vln.FineAmount) DESC) AS DenseRank,
    ROW_NUMBER() OVER (ORDER BY SUM(vln.FineAmount) DESC) AS RowNum,
    SUM(SUM(vln.FineAmount)) OVER () AS TotalAllFines,
    AVG(SUM(vln.FineAmount)) OVER () AS AvgFine
FROM TrafficViolations tv
INNER JOIN Vehicles v ON tv.VehicleID = v.VehicleID
INNER JOIN VehicleModels vmd ON v.ModelID = vmd.ModelID
INNER JOIN VehicleMakes vm ON vmd.MakeID = vm.MakeID
INNER JOIN Violations vln ON tv.ViolationID = vln.ViolationID
GROUP BY v.LicensePlate, vm.MakeName, vmd.ModelName
ORDER BY TotalFine DESC;
GO
